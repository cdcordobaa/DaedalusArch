## C3: Spec Parser — New Methods

```typescript
// Load a preset template by style name
loadPreset(styleName: string): DomainResult<ParsedSpec>
// Finds presets/{styleName}.yaml, parses it, returns typed spec

// Merge preset with project overrides
mergeSpecs(preset: ParsedSpec, overrides: Partial<SpecOverrides>): DomainResult<ParsedSpec>
// Project-specific values override preset defaults
// Layer directory mappings merged (project wins)
// Fitness functions: project can override threshold, enabled, exclude_paths

// List available presets
listPresets(): string[]
// Scans presets/ directory, returns style names

// Validate spec against project filesystem
validateSpec(spec: ParsedSpec, projectPath: string): DomainResult<ValidationReport>
// Checks: layer dirs exist, IDs unique, templates exist, thresholds valid, globs valid
// Returns structured errors with field paths and suggestions
```

### SpecOverrides Type
```typescript
interface SpecOverrides {
  layers?: LayerOverride[]           // Override directory mappings
  fitness_functions?: FFOverride[]   // Override thresholds, enabled, exclude_paths
  exclude_paths?: string[]           // Global exclude paths
  llm_config?: LLMConfig            // Gemini model/params
}

interface FFOverride {
  id: string
  threshold?: number
  enabled?: boolean
  reason?: string
  exclude_paths?: string[]
  severity?: Severity
}
```

## C7: LLM Critic — GeminiProvider Methods

```typescript
class GeminiProvider implements LLMProvider {
  constructor(config: GeminiConfig)

  // Execute LLM evaluation (matches existing LLMProvider interface)
  evaluate(prompt: string, options: LLMOptions): Promise<DomainResult<LLMResponse>>

  // Get configured model name
  getModelName(): string

  // Check if provider is available (API key configured)
  isAvailable(): boolean
}

interface GeminiConfig {
  apiKey: string
  model?: string          // default: "gemini-2.0-flash"
  temperature?: number    // default: 0
  maxTokens?: number      // default: 4096
}

// Factory method on existing LLMProviderFactory
class LLMProviderFactory {
  // Extended: "gemini" option added
  static create(provider: 'mock' | 'gemini', config?: LLMProviderConfig): LLMProvider
  // Returns NullLLMProvider when API key missing (graceful degradation)
}

class NullLLMProvider implements LLMProvider {
  evaluate(): Promise<DomainResult<LLMResponse>>
  // Returns DomainResult.error with "No LLM provider configured" — signals symbolic-only mode
  isAvailable(): boolean  // always false
}
```

## C12: Baseline Manager — All Methods

```typescript
class BaselineManager {
  // Create baseline from current violations
  createBaseline(violations: Violation[]): BaselineSnapshot
  // Generates stable keys, serializes to snapshot

  // Load existing baseline from file
  loadBaseline(filePath: string): DomainResult<BaselineSnapshot>
  // Reads JSON, validates structure, returns typed snapshot

  // Save baseline to file
  saveBaseline(snapshot: BaselineSnapshot, filePath: string): DomainResult<void>

  // Compare current violations against baseline
  compare(current: Violation[], baseline: BaselineSnapshot): BaselineResult
  // Returns: { baselineViolations, newViolations, removedFromBaseline }

  // Generate stable violation key for matching
  generateKey(violation: Violation): string
  // Hash of: ruleId + filePath + violationType + (line range or signature)
}

interface BaselineSnapshot {
  version: string
  createdAt: string          // ISO 8601
  specFile: string
  totalViolations: number
  violations: BaselineEntry[]
}

interface BaselineEntry {
  key: string                // Stable hash
  ruleId: string
  filePath: string
  severity: Severity
  description: string
}

interface BaselineResult {
  baselineViolations: Violation[]   // In baseline — informational
  newViolations: Violation[]        // Not in baseline — fail CI
  removedFromBaseline: BaselineEntry[]  // Were in baseline, now fixed
  baselineFilePath: string
}
```

## C13: Report Generator — All Methods

```typescript
class ReportGenerator {
  // Generate self-contained HTML report
  generate(input: ReportInput): DomainResult<ReportOutput>
  // Orchestrates: build graph data → build sections → inject into template → write file

  // Build Cytoscape.js graph data from APG
  buildGraphData(apgResult: APGResult, spec: ParsedSpec, violations: ActionableViolation[]): CytoscapeGraphData
  // Maps components to nodes (colored by layer), dependencies to edges, violations to red edges

  // Build dashboard sections data
  buildDashboardData(report: EvaluationReport, baseline: BaselineResult | null): DashboardData
  // Transforms evaluation data into the shape the HTML template expects

  // Render HTML from template + data
  renderHTML(template: string, data: DashboardData, graphData: CytoscapeGraphData): string
  // Injects JSON data into template, returns complete HTML string
}

interface ReportInput {
  evaluationReport: EvaluationReport
  parsedSpec: ParsedSpec
  apgResult: APGResult
  baselineResult?: BaselineResult
  projectName: string
  specFilePath: string
  outputPath: string
}

interface ReportOutput {
  htmlContent: string
  outputPath: string
  sections: string[]     // List of sections rendered
}

interface CytoscapeGraphData {
  nodes: CytoscapeNode[]
  edges: CytoscapeEdge[]
  layerColors: Record<string, string>
}

interface DashboardData {
  header: HeaderData
  ahsScore: AHSScoreData
  fitnessFunctions: FitnessFunctionCardData[]
  violations: ActionableViolation[]
  baseline: BaselineSplitData | null
  summary: SummaryStatsData
}
```
