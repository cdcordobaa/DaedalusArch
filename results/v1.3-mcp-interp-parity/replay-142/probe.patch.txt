import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';
import { createPipeline } from './src/pipeline/pipeline-factory.js';
import { MemoryGraphRepository } from './src/cypher-interp/repository.js';
import { REPLAY_TAPES } from './scripts/lib/loop/replay-tapes.js';
import { runReplay } from './scripts/lib/loop/replay.js';
import { mcpFeedback } from './scripts/lib/loop/replay-verify.js';

async function main(): Promise<void> {
  const base = fs.mkdtempSync(path.join(os.tmpdir(), 'v1-replay-'));
  const evalEnv: NodeJS.ProcessEnv = { NEO4J_URI: 'bolt://unused', NEO4J_USER: 'unused', NEO4J_PASSWORD: 'unused', APG_STORE_PATH: path.join(base, 'apg-store') };
  const evaluateDeps = { makeGraph: () => new MemoryGraphRepository(), makePipeline: (c: Parameters<typeof createPipeline>[0]) => createPipeline({ ...c, graphEngine: 'cypher-interp' }) };
  const r = await runReplay({ repoRoot: process.cwd(), outRoot: path.join(base, 'out'), harnessRoot: path.join(base, 'harness'), evaluatorEnv: evalEnv, evaluateDeps, tapes: REPLAY_TAPES });
  if (!r.success) throw new Error(r.errors.map((e) => `${e.code}: ${e.message}`).join('; '));
  const rows = mcpFeedback(r.data, REPLAY_TAPES);
  process.stdout.write(`${JSON.stringify(rows.map((m) => ({ task: m.taskId, step: m.step, tool: m.toolReported, full: m.fullBlocking, fix: m.fixApplied, unfollowed: m.unfollowed, errors: m.toolErrors })), null, 1)}\n`);
  process.stdout.write(`engine=${process.env.DAEDALUS_FAST_RULES_ENGINE ?? 'fast'} base=${base}\n`);
}
void main().catch((e: unknown) => { process.stderr.write(`${e instanceof Error ? e.stack ?? e.message : String(e)}\n`); process.exitCode = 2; });
